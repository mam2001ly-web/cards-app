<script>
// إعدادات الاتصال بمشروعك على Supabase
const supabaseUrl = 'https://oamhbeewsexdrayrmeda.supabase.co'
const supabaseKey = 'sb_publishable_Xp4wtYqVz3cVbI2FkZMzRw_HiaawjU2'
const _supabase = supabase.createClient(supabaseUrl, supabaseKey)

async function addCard() {
const cardNum = document.getElementById('cardNumber').value;
if (!cardNum) {
alert('الرجاء إدخال رقم الكرت');
return;
}

const { data, error } = await _supabase
.from('cards')
.insert([{ card_number: cardNum }]);

if (error) {
alert('خطأ أثناء الحفظ: ' + error.message);
} else {
alert('تم إضافة الكرت بنجاح!');
document.getElementById('cardNumber').value = '';
fetchCards();
}
}

async function fetchCards() {
const { data, error } = await _supabase.from('cards').select('*').order('id', { ascending: false });
const listDiv = document.getElementById('cardsList');

if (error) {
listDiv.innerHTML = 'خطأ في جلب البيانات';
return;
}

if (data.length === 0) {
listDiv.innerHTML = '<p>لا توجد كروت مضافة حالياً</p>';
return;
}

listDiv.innerHTML = data.map(c => `
<div class="card-item">
<span><b>${c.card_number}</b></span>
<span style="color: ${c.status === 'active' ? 'green' : 'red'};">${c.status}</span>
</div>
`).join('');
}

// جلب الكروت عند فتح الصفحة
fetchCards();
